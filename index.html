الفصل 31 — الدالة sum() وحساب أسرع زمن

function sum(freestyle, breaststroke, backstroke) { let time = freestyle + breaststroke + backstroke; return time; } let teamRed = sum(39.5, 60.5, 58.5); console.log(teamRed); let teamBlue = sum(40, 62, 58); console.log(teamBlue); let teamGreen = sum(41, 59.5, 60); console.log(teamGreen); console.log("The fastest time is" + Math.min(teamRed, teamBlue, teamGreen) );
teamRed
158.5
teamBlue
160
teamGreen
160.5
أقل زمن
158.5

الدالة تجمع الأزمنة الثلاثة وتُرجع الناتج باستخدام return. ثم تُستخدم Math.min() لاختيار أصغر قيمة.

الفصل 32 — عدّ قيم Yes وNo

let volunteerList = ["Yes", "No", "Yes", "Yes", "Yes", "Yes", "No", "No", "Yes", "Yes", "Yes", "No", "Yes", "No"]; let count = 0; for (let day of volunteerList) { if (day == "Yes") { count++; } } console.log(count);

عند البحث عن Yes تكون النتيجة 9. وعند استبدال الشرط بـ day == "No" تكون النتيجة 5.

let count = 0; for (let day of volunteerList) { if (day == "No") { count++; } } console.log(count);

الفصل 32 — الشروط المتداخلة: الطقس وعدد الأصدقاء

let weather = "sunny"; let friends = 10; if (weather == "sunny") { if (friends >= 10) { console.log("play baseball"); } else { console.log("play basketball"); } } else { console.log("read a book"); }

الطقس مشمس، وعدد الأصدقاء 10؛ لذلك يتحقق الشرطان وتكون النتيجة play baseball.

الفصل 32 — الشروط المتداخلة: نوع الساندويتش

let sandwichType = "savory"; let savory = "tuna"; if (sandwichType == "savory") { if (savory == "tuna") { console.log("Exchange with Cody"); } else { console.log("Eat ham sandwich"); } } else { console.log("Eat strawberry jam sandwich"); }

بما أن sandwichType تساوي savory وsavory تساوي tuna، فالنتيجة Exchange with Cody.
